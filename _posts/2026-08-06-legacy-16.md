> 用前缀树维护字符串集合，支持快速插入与查询。
{: .prompt-info }

## 模板作用

用前缀树维护字符串集合，支持快速插入与查询。

## 适用场景

大量字符串、前缀统计或单词出现次数查询时使用。

## 复杂度

总时间 O(所有字符串长度之和)。

## 使用提醒

字符集大小决定子节点数组大小；插入结束位置要单独记录计数。

## 代码

````cpp
#include<bits/stdc++.h>
using namespace std;

const int N = 1e5+9;

int son[N][26],cnt[N],idx; //son记录子节点的整体树关系，cnt记录终点个数，idx记录当前整体下标到哪了
int n;

void insert(char str[]){  // 每次开始的时候记录的是目前的根节点，操作后转到当前子节点
    int p = 0;
    for(int i = 0 ; str[i] ; i++){
        int u = str[i] - 'a';
        if(!son[p][u]) son[p][u] = ++idx;
        p = son[p][u];
    }
    cnt[p]++;
}

void query(char str[]){  // 每次开始的时候记录的是目前的根节点，操作后转到当前子节点
    int p = 0;
    for(int i = 0 ; str[i] ; i++){
        int u = str[i] - 'a';
        if(!son[p][u]){
            printf("%d\n",0);
            return;
        }
        p = son[p][u];
    }
    printf("%d\n",cnt[p]);
}

int main(){
    scanf("%d",&n);
    char str[N],op[2];
    while(n--){
        scanf("%s%s",op,str);
        if(op[0] == 'I') insert(str);
        else query(str);
    }
    return 0;
}
````
