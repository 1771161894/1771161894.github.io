> 统计 n! 中每个质数的指数。
{: .prompt-info }

## 模板作用

统计 n! 中每个质数的指数。

## 适用场景

组合数、整除性或阶乘因子相关题目时使用。

## 复杂度

约 O(n log log n + π(n) log n)。

## 使用提醒

先筛出所有质数，再累加 n / p、n / p² 等项。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 1e6+10;

int primes[N];
int cnt;
bool st[N];

int get_primes(int n){
    for(int i = 2 ; i <= n ; i++){
        if(!st[i]) primes[cnt ++] = i;
        for(int j = 0 ; primes[j] <= n / i; j++){
            st[primes[j] * i] = true;
            if(i % primes[j] == 0) break;
        }
    }
}

int main(){
    int n;
    scanf("%d",&n);
    get_primes(n);
    for(int i = 0 ; i < cnt ; i++){
        int p = primes[i];
        int s = 0 , t = n;
        while(t) s += t / p , t /= p;
        printf("%d %d\n",p,s);
    }
    return 0;
}
````
