> 通过行变换求解线性方程组，并判断无解或多解。
{: .prompt-info }

## 模板作用

通过行变换求解线性方程组，并判断无解或多解。

## 适用场景

未知数和方程数较小、需要处理实数线性方程组时使用。

## 复杂度

时间 O(n³)，空间 O(n²)。

## 使用提醒

浮点比较要使用 eps；换行和归一化后注意只消去需要的列。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 110;
const double eps = 1e-8;

int n;
double a[N][N];



int gauss(){
    int r,c;
    for(c = 0 , r = 0; c < n ; c++){
        int t = r;
        for(int i = r ; i < n ; i++){
            if(fabs(a[i][c]) > fabs(a[t][c]))
                t = i;
        }
        if(fabs(a[t][c]) < eps) continue;

        for(int i = c ; i <= n ; i ++) swap(a[t][i],a[r][i]); // 交换列主元最大的第r行

        for(int i = n ; i >= c ; i --) a[r][i] /= a[r][c]; // 本行的第一个元素归一化

        for(int i = r+1 ; i < n ; i++) // 把下面所有的列主元都弄成0
            for(int j = n ; j >= c ; j--)
                a[i][j] -= a[i][c] * a[r][j];

        r ++;
    }

    if(r < n){
        for(int i = r ; i < n ; i ++)
            if(fabs(a[i][n]) > eps)
                return -1;
        return 0;
    }else{
        for(int i = n-1 ; i >= 0 ; i --)
            for(int j = n-1 ; j >= i+1 ; j--)
                a[i][n] -= a[i][j] * a[j][n];
        return 1;
    }
}

int main(){
    scanf("%d",&n);
    for(int i = 0 ; i < n ; i++)
        for(int j = 0 ; j < n+1 ; j++)
            scanf("%lf",&a[i][j]);

    int t = gauss();
    if(t == -1)puts("No solution");
    else if(!t) puts("Infinite group solutions");
    else for(int i = 0 ; i < n ; i++) printf("%.2lf\n",a[i][n]);
    return 0;
}
````
