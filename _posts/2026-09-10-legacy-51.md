#include<bits/stdc++.h>

using namespace std;

const int N = 1e6+10;

string s, p;
int ne[N];

int main(){
    int n, m;
    cin >> n >> p >> m >> s;
    for(int i = 1, j = 0 ; i < n ; i ++){
        while(j && p[i] != p[j]) j = ne[j-1];
        if(p[i] == p[j]) j ++;
        ne[i] = j;
    }
    for(int i = 0, j = 0 ; i < m ; i ++){
        while(j && s[i] != p[j]) j = ne[j-1];
        if(s[i] == p[j]) j ++;
        if(j == n) {
            cout << i-j+1 << " ";
            j = ne[j-1];
        }
    }
    return 0;
}
