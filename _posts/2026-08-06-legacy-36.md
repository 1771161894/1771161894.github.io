> 维护前缀和，支持单点修改与前缀/区间求和。

## 模板作用

维护前缀和，支持单点修改与前缀/区间求和。

## 适用场景

动态数组求和、逆序对等在线问题时使用。

## 复杂度

单次修改和查询均为 O(log n)。

## 使用提醒

lowbit(x) = x & -x；下标必须从 1 开始，区间和由两个前缀和相减。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 1e5+10;

int a[N],tr[N];
int n,m;

int lowbit(int x){
    return x & -x;
}

void add(int u, int x){
    for(int i = u; i <= n; i += lowbit(i)) tr[i] += x;
}

int query(int u){
    int res = 0;
    for(int i = u ; i ; i -= lowbit(i)) res += tr[i];
    return res;
}

int main(){
    scanf("%d%d",&n,&m);
    for(int i = 1 ; i <= n ; i ++) scanf("%d",&a[i]);
    for(int i = 1 ; i <= n ; i ++) add(i,a[i]);
    while(m--){
        int op,x,y;
        scanf("%d%d%d",&op,&x,&y);
        if(op == 0) printf("%d\n",query(y)-query(x-1));
        else add(x,y);
    }
    return 0;
}
````
