> 借助小根堆每次取当前距离最小的点，求非负边权单源最短路。
{: .prompt-info }

## 模板作用

借助小根堆每次取当前距离最小的点，求非负边权单源最短路。

## 适用场景

稀疏图、点和边规模较大时优先使用。

## 复杂度

时间 O((n + m) log n)，空间 O(n + m)。

## 使用提醒

仅适用于非负边；出堆后先跳过已经确定最短路的旧状态。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 2e5+9;

typedef pair<int,int> PII;

int n,m;
int h[N],e[N],ne[N],idx;
int w[N],dist[N];
bool st[N];

void add(int a,int b,int c){
    e[idx] = b;
    w[idx] = c;
    ne[idx] = h[a];
    h[a] = idx++;
}

int dijkstra(){
    memset(dist,0x3f,sizeof dist);
    dist[1] = 0;
    priority_queue<PII,vector<PII>,greater<PII>> heap; // 堆里存的是当前dist[]值最小的点
    heap.push({0,1});
    while(heap.size()){
        auto t = heap.top();
        heap.pop();
        int vex = t.second,d = t.first;
        if(st[vex]) continue; //这里是在更新之前检验是否已经确定，以此保证重边问题自然解决
        st[vex] = true;
        for(int i = h[vex]; i != -1 ; i = ne[i]){
            int j = e[i];
            if(dist[j] > d+w[i]){
            dist[j] = d+w[i];
            heap.push({dist[j],j}); // 不管是不是最小的都入堆，堆自动排序
            }
        }
    }
    if(dist[n] == 0x3f3f3f3f) return -1;
    else return dist[n];
}

int main(){
    memset(h,-1,sizeof h);
    scanf("%d%d",&n,&m);
    while(m--){
        int a,b,c;
        scanf("%d%d%d",&a,&b,&c);
        add(a,b,c);
    }

    printf("%d",dijkstra());
    return 0;
}
````
