> 用开放寻址或拉链法快速判断整数是否出现过。

## 模板作用

用开放寻址或拉链法快速判断整数是否出现过。

## 适用场景

需要大量插入、查找且不方便用 STL 容器时使用。

## 复杂度

平均单次 O(1)。

## 使用提醒

哈希表容量要略大于数据规模；处理负数时先把模值调整为非负。

## 代码

````cpp
//开放寻址法实现哈希表

#include<bits/stdc++.h>
using namespace std;

const int N = 200003,null = 0x3f3f3f3f;

int h[N],n;

int find(int x){  //实现查询功能，返回应该存储的位置
    int k = (x % N + N) % N;
    while(h[k] != null && h[k] != x ){
        k++;
        if(k == N) k = 0;
    }
    return k;
}

int main(){
    scanf("%d",&n);
    memset(h,0x3f,sizeof(h));
    while(n--){
        char op[2];
        int a,idx;
        scanf("%s%d",op,&a);
        idx = find(a);
        if(op[0] == 'I') h[idx] = a;
        else{
            if(h[idx] != null) puts("Yes");
            else puts("No");
        }
    }
    return 0;
}


// 拉链法实现哈希表

#include<bits/stdc++.h>
using namespace std;

const int N = 100003;

int h[N],n;
int e[N],ne[N],idx;

void insert(int x){
    int k = (x % N + N) % N;
    e[idx] = x;
    ne[idx] = h[k];
    h[k] = idx ++;
}

bool find(int x){
    int k = (x % N + N) % N;
    for(int i = h[k] ; i != -1; i = ne[i]){
        if(e[i] == x) return true;
    }
    return false;
}

int main(){
    scanf("%d",&n);
    memset(h,-1,sizeof(h)); // 都赋值为-1代指当前所有的拉链头都是空的
    while(n--){
        char op[2];
        int x;
        scanf("%s%d",op,&x);
        if(op[0] == 'I') insert(x);
        else{
            if(find(x)) puts("Yes");
            else puts("No");
        }
    }
    return 0;
}
````
