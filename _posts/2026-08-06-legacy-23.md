> 用数组逐位相减并处理借位，计算两个超长非负整数之差。

## 模板作用

用数组逐位相减并处理借位，计算两个超长非负整数之差。

## 适用场景

大整数相减且已知被减数不小于减数时使用。

## 复杂度

时间 O(max(n, m))，空间 O(max(n, m))。

## 使用提醒

先比较大小；借位后的当前位要加 10，输出时清除前导零。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 1e5+10;

bool cmp(vector<int> &a,vector<int> &b){
    if(a.size() != b.size()){
        if(a.size() > b.size()) return true;
        else return false;
    }else{
        for(int i = a.size() - 1; i >= 0 ; i--){
            if(a[i] > b[i]) return true;
            else if(a[i] < b[i]) return false;
        }
    }
    return true;
}

vector<int> sub(vector<int> &a,vector<int> &b){
    int t = 0;
    vector<int> res;
    for(int i = 0 ; i < a.size() ; i ++){
        t += a[i];
        if(i < b.size()) t -= b[i];
        res.push_back((t+10)%10);
        if(t<0) t = -1;
        else t = 0;
    }
    while(res.size() > 1 && res.back() == 0) res.pop_back();
    return res;
}

int main(){
    char sa[N],sb[N];
    vector<int> a,b,res;
    scanf("%s%s",sa,sb);
    for(int i = strlen(sa) - 1; i >= 0 ; i--) a.push_back(sa[i] - '0');
    for(int i = strlen(sb) - 1; i >= 0 ; i--) b.push_back(sb[i] - '0');
    if(cmp(a,b)) res = sub(a,b);
    else printf("-"),res = sub(b,a);
    for(int i = res.size() -1 ; i >= 0 ; i--) printf("%d",res[i]);
    return 0;
}
````
