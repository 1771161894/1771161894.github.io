> 利用队列优先松弛可能变短的点，求单源最短路。

## 模板作用

利用队列优先松弛可能变短的点，求单源最短路。

## 适用场景

含负边但通常数据较温和时使用。

## 复杂度

最坏 O(nm)，平均表现常较好。

## 使用提醒

同一节点在队列中时不要重复入队；SPFA 在恶意数据下可能很慢。

## 代码

````cpp
// spfa 找单源最短路径
#include<bits/stdc++.h>

using namespace std;

const int N = 1e5+9;

int n,m;
int h[N],e[N],ne[N],idx;
int dist[N],w[N];
bool st[N];


void add(int a,int b,int c){
    e[idx] = b;
    w[idx] = c;
    ne[idx] = h[a];
    h[a] = idx ++;
}

void spfa(){
    memset(dist,0x3f,sizeof dist);
    dist[1] = 0;

    queue<int> q; // spfa算法的核心：用一个队列来维护所有做过改变的起点，然后用类似宽搜的方式来遍历出度，减少无用遍历次数
    q.push(1);
    st[1] = true;

    while(q.size()){
        auto t = q.front();
        q.pop();
        st[t] = false;
        for(int i = h[t] ; i != -1 ; i = ne[i]){
            int j = e[i];
            if(dist[j] > dist[t] + w[i]){
                dist[j] = dist[t] + w[i];
                if(!st[j]) q.push(j),st[j] = true;
            }
        }
    }
}

int main(){
    memset(h,-1,sizeof h);
    scanf("%d%d",&n,&m);
    while(m--){
        int a,b,c;
        scanf("%d%d%d",&a,&b,&c);
        add(a,b,c);
    }
    spfa();
    if(dist[n] == 0x3f3f3f3f) puts("impossible");
    else printf("%d\n",dist[n]);
}
````
