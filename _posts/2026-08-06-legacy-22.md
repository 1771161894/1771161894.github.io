> 用数组逐位相加，处理超长非负整数。

## 模板作用

用数组逐位相加，处理超长非负整数。

## 适用场景

数字长度超过 long long 时使用。

## 复杂度

时间 O(max(n, m))，空间 O(max(n, m))。

## 使用提醒

低位先算并维护进位；循环结束后不要漏掉最后一个进位。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 1e5+10;

vector<int> add(vector<int> &a,vector<int> &b){
    int t = 0;
    vector<int> res;
    for(int i = 0 ; i < a.size() || i < b.size() || t; i ++){
        if(i < a.size()) t += a[i];
        if(i < b.size()) t += b[i];
        res.push_back(t % 10);
        t /=10;
    }
    return res;
}

int main(){
    char sa[N],sb[N];
    vector<int> a,b,res;
    scanf("%s%s",sa,sb);
    for(int i = strlen(sa) - 1 ; i >= 0 ; i--) a.push_back(sa[i] - '0');
    for(int i = strlen(sb) - 1 ; i >= 0 ; i--) b.push_back(sb[i] - '0');
    res = add(a,b);
    for(int i = res.size() - 1 ; i >= 0 ; i--) printf("%d",res[i]);
    return 0;
}
````
