> 每次在线性扫描中选出当前距离最小的点，求非负边权单源最短路。

## 模板作用

每次在线性扫描中选出当前距离最小的点，求非负边权单源最短路。

## 适用场景

顶点数不大、图较稠密且使用邻接矩阵时使用。

## 复杂度

时间 O(n²)，空间 O(n²)。

## 使用提醒

仅适用于非负边；选点后用该点松弛所有邻点，未连通时结果为无穷大。

## 代码

````cpp
//朴素dijkstra，使用邻接矩阵适合稠密图

#include<bits/stdc++.h>

using namespace std;

const int N = 510;

int n,m;
int g[N][N]; // 邻接矩阵
int dist[N];
bool st[N]; // 是否已经找到最短边

int dijkstra(){
    memset(dist, 0x3f, sizeof dist);
    dist[1] = 0;
    for(int i = 1 ; i <= n ; i++){
        int t = -1; // 临时变量用来记录当前的最短dist
        for(int j = 1; j <= n ; j++)
            if(!st[j] && (t == -1 || dist[t] > dist[j]))
                t = j;

        st[t] = true;

        for(int j = 1; j <= n ; j++) // 使用新找的 t来 更新所有其他的边
            dist[j] = min(dist[j],dist[t]+g[t][j]);
    }
    if(dist[n] != 0x3f3f3f3f) return dist[n];
    else return -1;
}

int main(){
    memset(g, 0x3f, sizeof g);
    scanf("%d%d",&n,&m);
    while(m--){
        int a,b,w;
        scanf("%d%d%d",&a,&b,&w);
        g[a][b] = min(g[a][b],w);
    }
    printf("%d",dijkstra());
    return 0;
}
````
