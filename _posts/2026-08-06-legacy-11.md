> 按边权从小到大尝试加入边，并用并查集避免成环。

## 模板作用

按边权从小到大尝试加入边，并用并查集避免成环。

## 适用场景

边较多、需要构造最小生成树时使用。

## 复杂度

时间 O(m log m)，空间 O(n)。

## 使用提醒

最终选边数为 n - 1 才说明图连通；并查集下标范围要覆盖所有点。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 1e5+10, M = 2e5+10, INF = 0x3f3f3f3f;

struct Edge{
    int a,b,c;
    bool operator< (const Edge &W) const{ // 重载小于号方便直接使用sort进行快排
        return c < W.c;
    }
}edge[M];

int n,m;
int p[N];

int find(int x){
    if(p[x] != x) p[x] = find(p[x]);
    return p[x];
}

int kruskal(){
    sort(edge,edge + m);
    for(int i = 1 ; i <= n ; i ++) p[i] = i;
    int res = 0, cnt = 0;
    for(int i = 0 ; i < m ; i ++){
        int a = edge[i].a, b = edge[i].b, c = edge[i].c;
        a = find(a),b = find(b);
        if(a != b){
            p[a] = b;
            cnt ++;
            res += c;
        }
    }
    if(cnt < n-1) return INF;
    else return res;
}

int main(){
    scanf("%d%d",&n,&m);
    for(int i = 0 ; i < m ; i++){
        int a,b,c;
        scanf("%d%d%d",&a,&b,&c);
        edge[i] = {a,b,c};
    }

    int res = kruskal();

    if(res == INF) puts("impossible");
    else printf("%d",res);
    return 0;
}
````
