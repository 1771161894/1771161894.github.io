> 每次从未加入的点中选最小边权，逐步扩展最小生成树。
{: .prompt-info }

## 模板作用

每次从未加入的点中选最小边权，逐步扩展最小生成树。

## 适用场景

图较稠密、以邻接矩阵存图时使用。

## 复杂度

时间 O(n²)，空间 O(n²)。

## 使用提醒

应先判断当前最小点是否存在；不连通图无法得到完整最小生成树。

普通写法
```cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 510 , INF = 0x3f3f3f3f;

int n,m;
int g[N][N];
int dist[N];
bool st[N];

int prim(){ // 和dijkstra的代码很类似
    memset(dist,0x3f,sizeof dist);
    int res = 0;
    for(int i = 0 ; i < n ; i ++){
        int t = -1;
        for(int j = 1 ; j <= n ; j++)
            if(!st[j] && (t == -1 || dist[t] > dist[j]))
                t = j;
        if(i && dist[t] == INF) return INF;
        //下面的部分需要写到更新距离之前，避免因为自环导致出错
        if(i) res += dist[t]; // 第一次就不加距离进去res，其他都加入
        st[t] = true;
        for(int j = 1 ; j <= n ; j ++) dist[j] = min(dist[j],g[t][j]);
    }
    return res;
}

int main(){
    scanf("%d%d",&n,&m);
    memset(g,0x3f,sizeof g);
    while(m--){
        int a,b,c;
        scanf("%d%d%d",&a,&b,&c);
        g[a][b] = g[b][a] = min(g[a][b],c);
    }
    int res = prim();
    if(res == INF) puts("impossible");
    else printf("%d",res);
    return 0;
}
```

堆优化版本（建议记忆）
```cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 550, M = 200010;

typedef pair<int, int> pii;

int e[M], ne[M], w[M], idx, h[N];
int n,m;
bool st[N];
int dist[N];

void add(int a, int b, int c){
    e[idx] = b;
    ne[idx] = h[a];
    w[idx] = c;
    h[a] = idx ++;
}

struct cmp{
    bool operator() (const pii& a, const pii& b){
        return a.second > b.second;
    }
};

void prim(){
    int res = 0, cnt = 0;
    memset(dist, 0x3f, sizeof dist);
    dist[1] = 0;
    priority_queue<pii, vector<pii>, cmp> heap;
    heap.push({1, 0});
    while(!heap.empty()){
        auto t = heap.top();
        heap.pop();
        int vex = t.first, distance = t.second;
        if(st[vex]) continue;
        st[vex] = true;
        res += distance;
        cnt ++;
        for(int i = h[vex] ; i != -1 ; i = ne[i]){
            int j = e[i];
            if(!st[j] && dist[j] > w[i]){
                dist[j] = w[i];
                heap.push({j, dist[j]});
            }
        }
    }
    if(cnt == n) cout << res << endl;
    else puts("impossible");
    return;
}

int main(){
    cin >> n >> m;
    memset(h, -1, sizeof h);
    for(int i = 0 ; i < m ; i ++){
        int a, b, c;
        cin >> a >> b >> c;
        add(a, b, c);
        add(b, a, c);
    }
    prim();
    return 0;
}
```
