> 通过增广路求二分图最大匹配。

## 模板作用

通过增广路求二分图最大匹配。

## 适用场景

分配、配对、两类对象建立一对一关系时使用。

## 复杂度

时间 O(nm)，空间 O(n + m)。

## 使用提醒

每轮 DFS 都要重置访问标记；匹配数组保存右侧点当前匹配的左侧点。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 1e5+10 , M = 2e5 +10;

int n,m;
int h[N],e[M],ne[M],idx;
int color[N];

void add(int a,int b){
    e[idx] = b;
    ne[idx] = h[a];
    h[a] = idx++;
}

bool dfs(int u,int c){ // 实际上就是一个深搜的过程
    color[u] = c;
    for(int i = h[u] ; i != -1 ; i = ne[i]){
        int j = e[i];
        if(!color[j]){
            if(!dfs(j,3-c)) return false;
        }else if(color[j] == c) return false;
    }
    return true;
}

int main(){
    memset(h,-1,sizeof h);
    scanf("%d%d",&n,&m);
    while(m--){
        int a,b;
        scanf("%d%d",&a,&b);
        add(a,b),add(b,a);
    }

    bool flag = true;

    for(int i = 1 ; i <= n ; i++) // 避免有非连通图的情况
        if(!color[i])
            if(!dfs(i,1)){
                flag = false;
                break;
            }

    if(flag) puts("Yes");
    else puts("No");
    return 0;
}
````
