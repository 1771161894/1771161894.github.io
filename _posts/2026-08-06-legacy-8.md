> 最短路dijkstra算法模板
{: .prompt-info }

**原分类难度：** 入门

dijkstra的模板核心在于贪心选择，即每次找到离当前最近的点，然后加入已访问并且更新距离。
朴素版本为简单遍历邻接矩阵即可。（适用于稠密图）

```cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 510;

int n,m;
int g[N][N]; // 邻接矩阵
int dist[N];
bool st[N]; // 是否已经找到最短边

int dijkstra(){
    memset(dist, 0x3f, sizeof dist);
    dist[1] = 0;
    for(int i = 1 ; i <= n ; i++){
        int t = -1; // 临时变量用来记录当前的最短dist
        for(int j = 1; j <= n ; j++)
            if(!st[j] && (t == -1 || dist[t] > dist[j]))
                t = j;

        st[t] = true;

        for(int j = 1; j <= n ; j++) // 使用新找的 t来 更新所有其他的边
            dist[j] = min(dist[j],dist[t]+g[t][j]);
    }
    if(dist[n] != 0x3f3f3f3f) return dist[n];
    else return -1;
}

int main(){
    memset(g, 0x3f, sizeof g);
    scanf("%d%d",&n,&m);
    while(m--){
        int a,b,w;
        scanf("%d%d%d",&a,&b,&w);
        g[a][b] = min(g[a][b],w);
    }
    printf("%d",dijkstra());
    return 0;
}
```

堆优化版本为邻接表。（适用于稀疏图）
```cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 2e5+9;

typedef pair<int,int> PII;

int n,m;
int h[N],e[N],ne[N],idx;
int w[N],dist[N];
bool st[N];

void add(int a,int b,int c){
    e[idx] = b;
    w[idx] = c;
    ne[idx] = h[a];
    h[a] = idx++;
}

int dijkstra(){
    memset(dist,0x3f,sizeof dist);
    dist[1] = 0;
    priority_queue<PII,vector<PII>,greater<PII>> heap; // 堆里存的是当前dist[]值最小的点
    heap.push({0,1});
    while(heap.size()){
        auto t = heap.top();
        heap.pop();
        int vex = t.second,d = t.first;
        if(st[vex]) continue; //这里是在更新之前检验是否已经确定，以此保证重边问题自然解决
        st[vex] = true;
        for(int i = h[vex]; i != -1 ; i = ne[i]){
            int j = e[i];
            if(dist[j] > d+w[i]){
            dist[j] = d+w[i];
            heap.push({dist[j],j}); // 不管是不是最小的都入堆，堆自动排序
            }
        }
    }
    if(dist[n] == 0x3f3f3f3f) return -1;
    else return dist[n];
}

int main(){
    memset(h,-1,sizeof h);
    scanf("%d%d",&n,&m);
    while(m--){
        int a,b,c;
        scanf("%d%d%d",&a,&b,&c);
        add(a,b,c);
    }

    printf("%d",dijkstra());
    return 0;
}
```
