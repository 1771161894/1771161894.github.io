> 计算第 n 个卡特兰数，常用于合法括号和二叉树计数。
{: .prompt-info }

## 模板作用

计算第 n 个卡特兰数，常用于合法括号和二叉树计数。

## 适用场景

括号匹配、出栈序列、二叉搜索树形态等计数题时使用。

## 复杂度

取决于高精度乘除实现；常见写法约 O(n²)。

## 使用提醒

先确认题目对应的卡特兰模型；结果增长很快，通常需要高精度。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

typedef long long LL;

const int N = 2e5 + 10, mod = 1e9+7;

int fact[N],infact[N];

int qmi(int a,int b,int p){
    int res = 1;
    int t = a;
    while(b){
        if(b & 1) res = (LL) res * t % p;
        t = (LL) t * t % p;
        b >>= 1;
    }
    return res;
}

void init(){
    fact[0] = infact[0] = 1;
    for(int i = 1; i < N ; i ++){
        fact[i] = (LL) fact[i-1] * i % mod;
        infact[i] = (LL) infact[i-1] * qmi(i,mod - 2,mod) % mod;
    }
}

int main(){
    int n;
    init();
    scanf("%d",&n);
    int a =(LL) fact[2*n] * infact[n] % mod * infact[n] % mod;
    int b =(LL) fact[2*n] * infact[n-1] % mod * infact[n+1] % mod;
    int res = ((a-b) % mod + mod) % mod;
    printf("%d",res);
    return 0;
}
````
