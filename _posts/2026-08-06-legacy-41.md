> 根据质因数分解计算整数的全部约数之和。
{: .prompt-info }

## 模板作用

根据质因数分解计算整数的全部约数之和。

## 适用场景

约数相关的数论计数题时使用。

## 复杂度

单个数分解约 O(√n)。

## 使用提醒

每个质因子的贡献是 1 + p + … + p^a，多个质因子贡献相乘。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

const int N = 110, mod = 1e9+7;

typedef long long LL;

int n;
unordered_map<int,int> primes;
bool st[N];

void nums_divisors(int x){
    for(int i = 2 ; i <= x / i ; i++){
        while(x % i == 0) x /= i , primes[i] ++;
    }
    if(x > 1) primes[x] ++;
}

int main(){
    scanf("%d",&n);
    while(n--){
        int x;
        scanf("%d",&x);
        nums_divisors(x);
    }

    LL res = 1;

    for(auto t : primes){
        LL temp = 1;
        for(int i = 0 ; i < t.second ; i ++)    temp = (temp * t.first + 1) % mod;
        res = res * temp % mod;
    }

    printf("%d",res);
}
````
