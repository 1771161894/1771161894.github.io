> 在 SPFA 松弛过程中统计经过边数，检测是否存在负权环。
{: .prompt-info }

## 模板作用

在 SPFA 松弛过程中统计经过边数，检测是否存在负权环。

## 适用场景

需要判断图中是否含负环时使用。

## 复杂度

最坏 O(nm)，空间 O(n + m)。

## 使用提醒

通常把所有点入队或设置虚拟源点，避免漏掉不连通分量中的负环。

## 代码

````cpp
// spfa 判有无负环
#include<bits/stdc++.h>

using namespace std;

const int N = 10010;

int n,m;
int h[N],e[N],ne[N],w[N],idx;
int dist[N],cnt[N];
bool st[N];

void add(int a,int b,int c){
    e[idx] = b;
    w[idx] = c;
    ne[idx] = h[a];
    h[a] = idx ++;
}

bool spfa(){
    queue<int> q;

    for(int i = 1; i <= n ; i++) q.push(i),st[i] = true;
    /* 一开始要把所有的点都入队，这里看成有一个超级源点0，避免因为图不联通导致识别不出来。
    但是不需要一开始让所有的cnt[i] = 1 ，因为他实际上还是没有真正经过的，只是把结点入队而已 */
    while(q.size()){
        auto t = q.front();
        q.pop();
        st[t] = false;

        for(int i = h[t]; i != -1 ; i = ne[i]){
            int j = e[i];
            if(dist[j] > dist[t] + w[i]){
                dist[j] = dist[t] + w[i];
                cnt[j] = cnt[t] +1; // 到结点j为止一共走的跳数
                if(cnt[j] >= n) return true; // 如果总跳数 >= n ，那就说明至少有 n+1 个点，就说明有负环
                if(!st[j]) q.push(j),st[j] = true;
            }
        }
    }
    return false;
}

int main(){
    memset(h,-1,sizeof h);
    scanf("%d%d",&n,&m);
    while(m--){
        int a,b,c;
        scanf("%d%d%d",&a,&b,&c);
        add(a,b,c);
    }

    if(spfa()) puts("Yes");
    else puts("No");
    return 0;
}
````
