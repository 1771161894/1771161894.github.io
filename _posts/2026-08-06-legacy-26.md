> 用扩展欧几里得算法求 ax ≡ b (mod m) 的一组解。
{: .prompt-info }

## 模板作用

用扩展欧几里得算法求 ax ≡ b (mod m) 的一组解。

## 适用场景

同余方程、模意义下的除法或逆元问题时使用。

## 复杂度

时间 O(log min(a, m))。

## 使用提醒

只有 gcd(a, m) 能整除 b 才有解；最终答案要规范到正模范围。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

typedef long long LL;

const int N = 1e5+10;

int exgcd(int a,int b,int &x,int &y){
    if(!b){
        x = 1,y = 0;
        return a;
    }
    int d = exgcd(b,a%b,y,x);
    y -= a/b*x;
    return d;
}

int main(){
    int n;
    scanf("%d",&n);
    while(n--){
        int a,b,m;
        scanf("%d%d%d",&a,&b,&m);
        int x,y;
        int d = exgcd(a,m,x,y);
        if(b % d) puts("impossible");
        else {
            LL md = m/d;
            LL ans = (LL)(b/d) * x;
            ans = (ans % md + md) % md; // 输出最小非负数解
            printf("%lld\n", ans);

        } // printf("%lld\n",(LL)b/d * x % m); // 随便输出一个代表性的解，并且加了mod m防止溢出
    }
    return 0;
}
````
