> 预处理杨辉三角，快速查询小范围组合数。

## 模板作用

预处理杨辉三角，快速查询小范围组合数。

## 适用场景

n 范围较小、查询次数较多且不要求大模数技巧时使用。

## 复杂度

预处理 O(n²)，单次查询 O(1)。

## 使用提醒

初始化 C[i][0] = 1；数组上界要根据最大 n 预留。

## 代码

````cpp
#include<bits/stdc++.h>

using namespace std;

typedef long long LL;

const int N = 1e5+10,mod = 1e9+7;

int fact[N],infact[N];

LL qmi(int a,int b,int p){
    LL t = a;
    LL res = 1;
    while(b){
        if(b & 1) res =(LL) res * t % p;
        t =(LL) t * t % p;
        b >>= 1;
    }
    return res;
}

void init(){
    fact[0] = 1;
    infact[0] = 1;
    for(int i = 1 ; i < N ; i ++){
        fact[i] =(LL) fact[i-1] * i % mod;
        infact[i] =(LL) infact[i-1] * qmi(i,mod -2 , mod) % mod;
    }
}

int main(){
    int n ;
    scanf("%d",&n);
    init();
    while(n--){
        int a,b;
        scanf("%d%d",&a,&b);
        LL res;
        res = (LL) fact[a] * infact[b] % mod * infact[a-b] % mod;
        printf("%lld\n",res);
    }
    return 0;
}
````
